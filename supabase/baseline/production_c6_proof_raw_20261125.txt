-- ============================================================================
-- Phase C6 — Cohort Contribution Margin (read-only analytics)
-- File: supabase/migrations/20261125000001_phase_c6_cohort_contribution.sql
--
-- Managerial unit-economics view per (cohort, currency), as of a Damascus
-- business-day cutoff:
--   earned_revenue      = SUM(C5 earned_revenue) grouped by (cohort, currency)
--   trainer_cost        = SUM(trainer_dues.amount) where status <> 'cancelled'
--                         and period_end <= p_as_of (months ended by as-of)
--   contribution_margin = earned_revenue - trainer_cost
--   margin_ratio        = margin / earned, NULL when earned = 0 (unbounded)
--   trainees_count      = COUNT(DISTINCT trainee_id) over as-of enrollments
--   has_trainer_cost    = EXISTS a non-cancelled due row with period_end <= p_as_of
--
-- This is NOT net profit per cohort: shared/administrative expenses are not
-- attributable to a cohort in the current schema (expenses has no cohort_id),
-- and C3 trainer dues are computed on monthly cash collections after close.
-- ============================================================================
CREATE OR REPLACE FUNCTION public.get_cohort_contribution_report(
  p_as_of date
) RETURNS TABLE (
  cohort_id uuid,
  cohort_name text,
  course_name text,
  currency text,
  earned_revenue numeric,
  trainer_cost numeric,
  contribution_margin numeric,
  margin_ratio numeric,
  trainees_count int,
  has_trainer_cost boolean
)
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public, pg_temp
AS $$
#variable_conflict use_column
BEGIN
  -- 1. Security guard: Admin only
  IF NOT public.c1_is_admin() THEN
    RAISE EXCEPTION 'غير مصرّح: تقرير هامش مساهمة الأفواج متاح لمدير المركز فقط'
      USING ERRCODE = 'P0001';
  END IF;

  IF p_as_of IS NULL THEN
    RAISE EXCEPTION 'تاريخ التقرير مطلوب'
      USING ERRCODE = 'P0002';
  END IF;

  RETURN QUERY
  WITH revenue AS (
    SELECT
      r.cohort_id,
      max(r.cohort_name) AS cohort_name,
      max(r.course_name) AS course_name,
      r.currency,
      sum(r.earned_revenue)::numeric AS earned_revenue,
      count(DISTINCT r.trainee_id)::int AS trainees_count
    FROM public.get_accrual_revenue_report(p_as_of) AS r
    GROUP BY r.cohort_id, r.currency
  ),
  trainer AS (
    SELECT
      d.cohort_id,
      co.name AS cohort_name,
      c.name AS course_name,
      d.currency::text AS currency,
      sum(d.amount)::numeric AS trainer_cost
    FROM public.trainer_dues AS d
    JOIN public.cohorts AS co ON co.id = d.cohort_id
    JOIN public.courses AS c ON c.id = co.course_id
    WHERE d.status <> 'cancelled'
      AND d.period_end <= p_as_of
    GROUP BY d.cohort_id, co.name, c.name, d.currency
  ),
  merged AS (
    SELECT
      coalesce(rv.cohort_id, tr.cohort_id) AS cohort_id,
      coalesce(rv.cohort_name, tr.cohort_name) AS cohort_name,
      coalesce(rv.course_name, tr.course_name) AS course_name,
      coalesce(rv.currency, tr.currency) AS currency,
      coalesce(rv.earned_revenue, 0.00) AS earned_revenue,
      coalesce(tr.trainer_cost, 0.00) AS trainer_cost,
      coalesce(rv.trainees_count, 0) AS trainees_count,
      (tr.cohort_id IS NOT NULL) AS has_trainer_cost
    FROM revenue AS rv
    FULL OUTER JOIN trainer AS tr
      ON tr.cohort_id = rv.cohort_id
     AND tr.currency = rv.currency
  )
  SELECT
    m.cohort_id,
    m.cohort_name,
    m.course_name,
    m.currency,
    m.earned_revenue,
    m.trainer_cost,
    (m.earned_revenue - m.trainer_cost) AS contribution_margin,
    CASE
      WHEN m.earned_revenue > 0
        THEN (m.earned_revenue - m.trainer_cost) / m.earned_revenue
      ELSE NULL::numeric
    END AS margin_ratio,
    m.trainees_count,
    m.has_trainer_cost
  FROM merged AS m
  ORDER BY m.currency, m.cohort_name;
END;
$$;

-- Rebuild Validation Script (#47)
--
-- Compares database and platform state against the authoritative baseline.
-- Checks both core public schema invariants and external platform dependencies.
-- Safe, read-only verification that returns a summary table of all checks.
-- ============================================================================
CREATE OR REPLACE FUNCTION pg_temp.validate_database_rebuild()
RETURNS TABLE (
  check_id text,
  metric text,
  expected text,
  actual text,
  status text
)
LANGUAGE plpgsql
AS $val$
DECLARE
  v_tables integer;
  v_views integer;
  v_functions integer;
  v_policies integer;
  v_triggers integer;
  v_indexes integer;
  v_sequences integer;
  v_enums integer;
  v_rls_enabled integer;
  v_rls_forced integer;
  v_fingerprint text;
  v_unpinned integer;
  -- Platform metrics
  v_storage_bucket boolean := false;
  v_storage_policies integer := 0;
  v_cron_jobs integer := 0;
  v_extensions integer := 0;
  v_vault_secrets integer := 0;
BEGIN
  -- 1. Gather public schema counts
  SELECT count(*) INTO v_tables FROM pg_tables WHERE schemaname = 'public';
  SELECT count(*) INTO v_views FROM pg_views WHERE schemaname = 'public';
  SELECT count(*) INTO v_functions FROM pg_proc p JOIN pg_namespace n ON n.oid = p.pronamespace WHERE n.nspname = 'public';
  SELECT count(*) INTO v_policies FROM pg_policies WHERE schemaname = 'public';
  SELECT count(*) INTO v_triggers FROM pg_trigger t JOIN pg_class c ON c.oid = t.tgrelid JOIN pg_namespace n ON n.oid = c.relnamespace WHERE n.nspname = 'public' AND NOT t.tgisinternal;
  SELECT count(*) INTO v_indexes FROM pg_indexes WHERE schemaname = 'public';
  SELECT count(*) INTO v_sequences FROM pg_sequences WHERE schemaname = 'public';
  SELECT count(*) INTO v_enums FROM pg_type t JOIN pg_namespace n ON n.oid = t.typnamespace WHERE n.nspname = 'public' AND t.typtype = 'e';
  SELECT count(*) INTO v_rls_enabled FROM pg_class c JOIN pg_namespace n ON n.oid = c.relnamespace WHERE n.nspname = 'public' AND c.relkind = 'r' AND c.relrowsecurity;
  SELECT count(*) INTO v_rls_forced FROM pg_class c JOIN pg_namespace n ON n.oid = c.relnamespace WHERE n.nspname = 'public' AND c.relkind = 'r' AND c.relforcerowsecurity;

  -- 2. Compute public policy fingerprint
  SELECT md5(string_agg(
           schemaname || '.' || tablename || ':' || policyname || ':' ||
           cmd || ':' || roles::text || ':' || COALESCE(qual, '') || ':' ||
           COALESCE(with_check, ''),
           '|' ORDER BY schemaname, tablename, policyname
         )) INTO v_fingerprint
  FROM pg_policies
  WHERE schemaname = 'public';

  SELECT count(*) INTO v_unpinned
  FROM pg_proc p
  JOIN pg_namespace n ON n.oid = p.pronamespace
  WHERE n.nspname = 'public'
    AND p.prosecdef
    AND NOT EXISTS (
      SELECT 1 FROM unnest(COALESCE(p.proconfig, '{}')) AS c
      WHERE c LIKE 'search_path=%'
    );

  -- 4. Check platform dependencies
  IF to_regclass('storage.buckets') IS NOT NULL THEN
    SELECT EXISTS (SELECT 1 FROM storage.buckets WHERE id = 'course-materials') INTO v_storage_bucket;
  END IF;

  SELECT count(*) INTO v_storage_policies
  FROM pg_policies
  WHERE schemaname = 'storage' AND tablename = 'objects';

  IF to_regclass('cron.job') IS NOT NULL THEN
    SELECT count(DISTINCT command) INTO v_cron_jobs
    FROM cron.job
    WHERE active = true
      AND command IN (
        'SELECT public.auto_open_close_sessions(false);',
        'SELECT public.enqueue_session_reminders();',
        'SELECT public.dispatch_push_batch();',
        'SELECT public.expire_waitlist_notifications();',
        'SELECT public.enqueue_payment_reminders();'
      );
  END IF;

  SELECT count(*) INTO v_extensions
  FROM pg_extension
  WHERE extname IN ('uuid-ossp', 'pgcrypto', 'pg_stat_statements', 'pg_net', 'pg_cron', 'supabase_vault');

  IF to_regclass('vault.decrypted_secrets') IS NOT NULL THEN
    SELECT count(*) INTO v_vault_secrets
    FROM vault.decrypted_secrets
    WHERE name IN ('project_url', 'cron_secret', 'push_function_slug');
  END IF;

  -- Return rows
  RETURN QUERY VALUES
    ('C01', 'Tables in public', '43', v_tables::text, CASE WHEN v_tables = 43 THEN 'PASS' ELSE 'FAIL' END),
    ('C02', 'Views in public', '33', v_views::text, CASE WHEN v_views = 33 THEN 'PASS' ELSE 'FAIL' END),
    ('C03', 'Functions in public', '115', v_functions::text, CASE WHEN v_functions = 115 THEN 'PASS' ELSE 'FAIL' END),
    ('C04', 'Policies in public', '71', v_policies::text, CASE WHEN v_policies = 71 THEN 'PASS' ELSE 'FAIL' END),
    ('C05', 'Triggers in public', '68', v_triggers::text, CASE WHEN v_triggers = 68 THEN 'PASS' ELSE 'FAIL' END),
    ('C06', 'Indexes in public', '131', v_indexes::text, CASE WHEN v_indexes = 131 THEN 'PASS' ELSE 'FAIL' END),
    ('C07', 'Sequences in public', '3', v_sequences::text, CASE WHEN v_sequences = 3 THEN 'PASS' ELSE 'FAIL' END),
    ('C08', 'Enums in public', '9', v_enums::text, CASE WHEN v_enums = 9 THEN 'PASS' ELSE 'FAIL' END),
    ('C09', 'RLS Enabled Tables', '43', v_rls_enabled::text, CASE WHEN v_rls_enabled = 43 THEN 'PASS' ELSE 'FAIL' END),
    ('C10', 'RLS Forced Tables', '0', v_rls_forced::text, CASE WHEN v_rls_forced = 0 THEN 'PASS' ELSE 'FAIL' END),
    ('C11', 'Unpinned SecDef Functions', '0', v_unpinned::text, CASE WHEN v_unpinned = 0 THEN 'PASS' ELSE 'FAIL' END),
    ('C12', 'Policy MD5 Fingerprint', 'c8215afa7997fb5258254047a879d68a', v_fingerprint, CASE WHEN v_fingerprint = 'c8215afa7997fb5258254047a879d68a' THEN 'PASS' ELSE 'FAIL' END),
    ('C13', 'Storage Bucket (course-materials)', 'true', COALESCE(v_storage_bucket, false)::text, CASE WHEN COALESCE(v_storage_bucket, false) THEN 'PASS' ELSE 'FAIL' END),
    ('C14', 'Storage Policies on objects', '2', v_storage_policies::text, CASE WHEN v_storage_policies = 2 THEN 'PASS' ELSE 'FAIL' END),
